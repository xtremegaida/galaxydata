SELECT c.id, c.name, c.city, c.credit_limit, c.created, c.region_code, c.uid, c.notes
FROM (
   SELECT TOP (1) o.customer_id, o.id AS sort_key
   FROM orders AS o
   WHERE o.status = @p0
   ORDER BY o.id
) AS d
INNER JOIN customers AS c ON d.customer_id = c.id
ORDER BY d.sort_key
-- p0 string(20,ansi) = 'open'

### sales.customers.select(name, latest: orders_by_customer.orderBy(desc(order_date)).first())
SELECT c.name, (
   SELECT TOP (1) o2.status
   FROM orders AS o2
   WHERE o2.customer_id = c.id
   ORDER BY o2.order_date DESC, o2.id
) AS latest, c.id, (
   SELECT TOP (1) o.id
   FROM orders AS o
   WHERE o.customer_id = c.id
   ORDER BY o.order_date DESC, o.id
) AS id
FROM customers AS c

### sales.customers.select(name, last_total: orders_by_customer.orderBy(desc(order_date)).firstOrDefault().total, last_city: orders_by_customer.orderBy(desc(order_date)).first().ship_customer.city)
SELECT c.name, (
   SELECT TOP (1) o.total
   FROM orders AS o
   WHERE o.customer_id = c.id
   ORDER BY o.order_date DESC, o.id
) AS last_total, (
   SELECT c2.city
   FROM (
      SELECT TOP (1) o2.ship_customer_id, o2.order_date AS sort_key, o2.id AS sort_key_2
      FROM orders AS o2
      WHERE o2.customer_id = c.id
      ORDER BY o2.order_date DESC, o2.id
   ) AS d
   LEFT JOIN customers AS c2 ON d.ship_customer_id = c2.id
) AS last_city, c.id
FROM customers AS c

### sales.customers.where(orders_by_customer.orderBy(desc(order_date)).firstOrDefault() == null).select(name)
SELECT c.name, c.id
FROM customers AS c
WHERE NOT EXISTS (
   SELECT 1
   FROM orders AS o
   WHERE o.customer_id = c.id
)

### sales.customers.select(name, lines: orders_by_customer.orderBy(id).first().order_lines.count())
SELECT c.name, (
   SELECT COUNT(*) AS count
   FROM order_lines AS o
   WHERE o.order_id = (
      SELECT TOP (1) o2.id
      FROM orders AS o2
      WHERE o2.customer_id = c.id
      ORDER BY o2.id
   )
) AS lines, c.id
FROM customers AS c

### sales.orders.select(id, c: customer).select(id, c)

FROM orders AS o
INNER JOIN customers AS c ON o.customer_id = c.id

### g := sales.orders.where(total > 3); g.groupBy(o => o.customer_id).select(k: key, n: count())
SELECT o.customer_id AS k, COUNT(*) AS n
FROM orders AS o
WHERE o.total > @p0
GROUP BY o.customer_id
-- p0 decimal(10,2) = 3

### sales.orders.groupBy(customer, big: total > 100).select(customer, big, n: count(), s: sum(total))
SELECT d.name AS customer, d.big, COUNT(*) AS n, COALESCE(SUM(d.total), 0) AS s, d.id
FROM (
   SELECT o.total, c.name, o.customer_id AS id, CASE WHEN o.total > @p0 THEN CAST(1 AS bit) ELSE CAST(0 AS bit) END AS big
   FROM orders AS o
   INNER JOIN customers AS c ON o.customer_id = c.id
) AS d
GROUP BY d.id, d.big, d.name
-- p0 decimal(10,2) = 100

### sales.order_lines.select(order_id, qty, product)
SELECT o.order_id, o.qty, p.name AS product, o.line_no, p.code
FROM order_lines AS o
INNER JOIN products AS p ON o.product_code = p.code

### sales.orders.select(id, status).distinct()
SELECT DISTINCT o.id, o.status
FROM orders AS o

### sales.customers.select(name, t: orders_by_customer.orderBy(id).first().total).groupBy(name).select(name, s: sum(t))
SELECT d.name, COALESCE(SUM(d.sum_value), 0) AS s
FROM (
   SELECT c.name, (
      SELECT TOP (1) o.total
      FROM orders AS o
      WHERE o.customer_id = c.id
      ORDER BY o.id
   ) AS sum_value
   FROM customers AS c
) AS d
GROUP BY d.name

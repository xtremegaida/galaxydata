-- shop (SQLite)
SELECT o.id, o.total, c.name AS who, c.city
FROM orders AS o
LEFT JOIN customers AS c ON o.customer_id = c.id
WHERE o.status = @p0
ORDER BY o.id
-- @p0 string = 'open'

-- shop (DuckDB)
SELECT o.line_no, c.city
FROM order_lines AS o
INNER JOIN orders AS o2 ON o.order_id = o2.id
INNER JOIN customers AS c ON o2.customer_id = c.id
LIMIT 3

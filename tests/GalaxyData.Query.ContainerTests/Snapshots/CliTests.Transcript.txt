### the SQL of a page of groups, on SQL Server
exit 0
--- out
-- shop (SQL Server)
SELECT c.name, COUNT_BIG(*) AS n, COALESCE(SUM(o.total), 0) AS total, o.customer_id AS id
FROM orders AS o
INNER JOIN customers AS c ON o.customer_id = c.id
GROUP BY o.customer_id, c.name
ORDER BY COALESCE(SUM(o.total), 0) DESC
OFFSET 1 ROWS FETCH NEXT 1 ROWS ONLY
